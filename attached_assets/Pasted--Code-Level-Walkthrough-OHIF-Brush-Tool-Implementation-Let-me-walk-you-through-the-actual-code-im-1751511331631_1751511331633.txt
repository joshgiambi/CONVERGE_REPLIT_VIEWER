## **3. Command Execution Flow**

### **A. Tool Activation Command**
```typescript
// extensions/cornerstone/src/commandsModule.ts
setToolActive: ({ toolName, toolGroupId = null }) => {
  const { viewports } = viewportGridService.getState();
  
  if (!viewports.size) return;
  
  const toolGroup = toolGroupService.getToolGroup(toolGroupId);
  if (!toolGroup || !toolGroup?.hasTool(toolName)) return;
  
  // Deactivate current active tool
  const activeToolName = toolGroup.getActivePrimaryMouseButtonTool();
  if (activeToolName) {
    const activeToolOptions = toolGroup.getToolConfiguration(activeToolName);
    activeToolOptions?.disableOnPassive
      ? toolGroup.setToolDisabled(activeToolName)
      : toolGroup.setToolPassive(activeToolName);
  }
  
  // Activate new tool with mouse binding
  toolGroup.setToolActive(toolName, {
    bindings: [
      {
        mouseButton: Enums.MouseBindings.Primary,  // ← Left mouse button
      },
    ],
  });
}
```

### **B. Brush Size Management**

// extensions/cornerstone/src/commandsModule.ts
setBrushSize: ({ value, toolNames }) => {
  const brushSize = Number(value);
  
  toolGroupService.getToolGroupIds()?.forEach(toolGroupId => {
    if (toolNames?.length === 0) {
      // Update all brush tools in the tool group
      segmentationUtils.setBrushSizeForToolGroup(toolGroupId, brushSize);
    } else {
      // Update specific brush tools
      toolNames?.forEach(toolName => {
        segmentationUtils.setBrushSizeForToolGroup(toolGroupId, brushSize, toolName);
      });
    }
  });
}

class BrushTool extends BaseTool {
  static toolName = 'Brush';
  
  // Configuration determines brush behavior
  configuration = {
    activeStrategy: 'FILL_INSIDE_CIRCLE',  // or 'FILL_INSIDE_SPHERE'
    brushSize: 25,  // mm
    preview: {
      enabled: true,
    },

  // Event handlers
  mouseDownCallback = (evt) => {
    // 1. Get mouse position in image coordinates

    // 3. Update labelmap
    this._updateLabelmap(brushMask, currentImageIdIndex);
    
    // 4. Trigger render
    element.viewport.render();
  };
  
  mouseDragCallback = (evt) => {
    // Similar to mouseDown but for continuous drawing
    // Interpolates between previous and current mouse positions
  };
  
  mouseUpCallback = (evt) => {
    // Finalize the brush stroke
    // Commit changes to segmentation service
  };
}

### **B. Brush Mask Calculation**
```typescript
_calculateBrushMask(mousePos) {
  const { brushSize, activeStrategy } = this.configuration;
  
  if (activeStrategy === 'FILL_INSIDE_CIRCLE') {
    // 2D circle mask
    return this._createCircleMask(mousePos, brushSize);
  } else if (activeStrategy === 'FILL_INSIDE_SPHERE') {
    // 3D sphere mask
    return this._createSphereMask(mousePos, brushSize);
  }
}

_createCircleMask(center, radius) {
  // Convert radius from mm to pixels
  const radiusInPixels = this._mmToPixels(radius);
  
  // Create circular mask using distance calculation
  const mask = [];
  for (let y = -radiusInPixels; y <= radiusInPixels; y++) {
    for (let x = -radiusInPixels; x <= radiusInPixels; x++) {
      const distance = Math.sqrt(x*x + y*y);
      if (distance <= radiusInPixels) {
        mask.push({
          x: center.x + x,
          y: center.y + y,
          z: center.z
        });
      }
    }
  }
  return mask;
}
```

### **C. Labelmap Update Process**
```typescript
_updateLabelmap(brushMask, imageIndex) {
  // 1. Get active segmentation
  const segmentationService = this.servicesManager.services.segmentationService;
  const activeSegmentation = segmentationService.getActiveSegmentation();
  
  if (!activeSegmentation) {
    console.warn('No active segmentation for brush tool');
    return;
  }
  
  // 2. Get labelmap volume
  const labelmapVolume = activeSegmentation.representationData.Labelmap;
  
  // 3. Update voxels in the brush mask
  brushMask.forEach(voxel => {
    const { x, y, z } = voxel;
    
    // Convert to volume coordinates
    const volumeIndex = labelmapVolume.imageData.worldToIndex([x, y, z]);
    
    // Set the segment label (usually 1 for first segment)
    const segmentIndex = this._getActiveSegmentIndex();
    labelmapVolume.setPixelAt(volumeIndex, segmentIndex);
  });
  
  // 4. Notify segmentation service of changes
  segmentationService.updateSegmentation(activeSegmentation.segmentationId);
}
```

## **5. Segmentation Service Integration**

### **A. Segmentation State Management**
```typescript
// @ohif/core (segmentation service)
class SegmentationService {
  private segmentations = new Map();
  
  getActiveSegmentation(viewportId) {
    // Returns the currently active segmentation for the viewport
    return this.activeSegmentations.get(viewportId);
  }
  
  updateSegmentation(segmentationId) {
    // Triggers re-render of all viewports showing this segmentation
    this._notifyViewports(segmentationId);
  }
  
  addSegmentation(segmentation) {
    // Creates new labelmap volume and stores segmentation metadata
    const labelmapVolume = this._createLabelmapVolume(segmentation);
    this.segmentations.set(segmentation.segmentationId, {
      ...segmentation,
      representationData: {
        Labelmap: labelmapVolume
      }
    });
  }
}
```

### **B. Labelmap Volume Creation**
```typescript
_createLabelmapVolume(segmentation) {
  // Create 3D array with same dimensions as image volume
  const { dimensions, spacing, origin, direction } = this.imageVolume;
  
  const labelmapVolume = {
    volumeId: `${segmentation.segmentationId}-labelmap`,
    dimensions,
    spacing,
    origin,
    direction,
    imageData: new Uint8Array(dimensions[0] * dimensions[1] * dimensions[2]),
    
    // Methods for voxel access
    setPixelAt: (index, value) => {
      this.imageData[index] = value;
    },
    
    getPixelAt: (index) => {
      return this.imageData[index];
    }
  };
  
  return labelmapVolume;
}
```

## **6. Rendering Pipeline**

### **A. Segmentation Rendering**
```typescript
// Cornerstone rendering engine
class SegmentationRenderingEngine {
  renderSegmentation(viewport, segmentation) {
    const labelmapVolume = segmentation.representationData.Labelmap;
    const imageVolume = viewport.getImageData();
    
    // Create overlay canvas
    const overlayCanvas = this._createOverlayCanvas();
    
    // For each slice
    for (let z = 0; z < labelmapVolume.dimensions[2]; z++) {
      this._renderSlice(labelmapVolume, imageVolume, z, overlayCanvas);
    }
    
    // Blend overlay with image
    this._blendOverlay(viewport.canvas, overlayCanvas);
  }
  
  _renderSlice(labelmapVolume, imageVolume, sliceIndex, canvas) {
    // Extract 2D slice from 3D labelmap
    const sliceData = this._extractSlice(labelmapVolume, sliceIndex);
    
    // Map segment labels to colors
    const coloredSlice = this._applyColormap(sliceData);
    
    // Draw to overlay canvas
    this._drawToCanvas(coloredSlice, canvas, sliceIndex);
  }
}
```

This walkthrough shows how the brush tool integrates with OHIF's architecture, from user interaction through to labelmap updates and rendering. The key insight is that the brush tool operates on **labelmap volumes** (3D arrays of segment labels) rather than directly on image data, and the segmentation service manages the state and rendering of these labelmaps.
